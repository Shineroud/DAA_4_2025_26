class Solution {
public:
    int minEatingSpeed(vector<int>& piles, int h) {
        int lo=1;
        int hi=0;
        for(int p:piles) hi=max(hi, p);
        int ans=hi;
        while (lo<=hi) {
            int mid=lo+(hi-lo)/2;
            if(Finish(piles,h,mid)) {
                ans =mid;     
                hi=mid- 1;  
            }else{
                lo = mid + 1;  
            }
        }
        return ans;
    }
private:
    bool Finish(const vector<int>& piles, int h, int k) {
        long long totalHours = 0;
        for (int p : piles) {
            totalHours += (p + k - 1) / k;
        }
        return totalHours <= h;
    }
};
